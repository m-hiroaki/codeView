#include <sys/resource.h>
#include <libproc.h>
#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>
#include <stdint.h>
#include <string.h>
#include <time.h>

int main(int argc, char **argv) {
    if (argc != 3 || (strcmp(argv[2], "normal") && strcmp(argv[2], "nocache"))) return 2;
    int fd = open(argv[1], O_RDONLY);
    if (fd < 0) { perror("open"); return 1; }
    if (!strcmp(argv[2], "nocache") && fcntl(fd, F_NOCACHE, 1) != 0) { perror("F_NOCACHE"); close(fd); return 1; }
    struct rusage_info_v2 before = {0}, after = {0};
    if (proc_pid_rusage(getpid(), RUSAGE_INFO_V2, (rusage_info_t *)&before)) { perror("rusage before"); close(fd); return 1; }
    struct timespec start, stop;
    clock_gettime(CLOCK_MONOTONIC, &start);
    unsigned char buffer[65536];
    uint64_t total = 0;
    ssize_t count;
    while ((count = read(fd, buffer, sizeof(buffer))) > 0) total += (uint64_t)count;
    clock_gettime(CLOCK_MONOTONIC, &stop);
    if (count < 0) { perror("read"); close(fd); return 1; }
    if (proc_pid_rusage(getpid(), RUSAGE_INFO_V2, (rusage_info_t *)&after)) { perror("rusage after"); close(fd); return 1; }
    close(fd);
    if (after.ri_diskio_bytesread < before.ri_diskio_bytesread) return 1;
    double milliseconds = (stop.tv_sec - start.tv_sec) * 1000.0 + (stop.tv_nsec - start.tv_nsec) / 1000000.0;
    printf("{\"mode\":\"%s\",\"logicalReadBytes\":%llu,\"diskReadBefore\":%llu,\"diskReadAfter\":%llu,\"diskReadDelta\":%llu,\"milliseconds\":%.6f}\n",
        argv[2], (unsigned long long)total, (unsigned long long)before.ri_diskio_bytesread,
        (unsigned long long)after.ri_diskio_bytesread, (unsigned long long)(after.ri_diskio_bytesread - before.ri_diskio_bytesread), milliseconds);
    return 0;
}
